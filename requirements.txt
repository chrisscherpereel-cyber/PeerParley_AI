# PeerParley — what the app needs to run.
#
#   pip install -r requirements.txt
#
# Streamlit Community Cloud installs this file automatically on deploy.
# Needs Python 3.10 or newer.
#
# Every entry has a lower bound (the oldest version this code is written
# against) and an upper bound at the next major version. The upper bounds are
# the point: without them, a major release of pandas or Streamlit lands in a
# *fresh* install months from now and breaks it for a new user who changed
# nothing. Raise them deliberately, after running the tests.
#
# Optional extras — the AI provider SDKs — live in
# requirements-optional.txt. The app runs without any of them and says which
# package is missing if you pick a feature that needs one.

streamlit>=1.36,<2
pandas>=2.2,<4
numpy>=1.26,<3
openpyxl>=3.1,<4            # Excel contact lists
reportlab>=4.1,<6           # the PDF reports
cryptography>=42.0,<60      # encrypts every stored student record
requests>=2.32,<3           # talking to your storage provider

# The storage backends. These are NOT optional in practice: which one you use
# is a line in your secrets, and Streamlit Community Cloud installs only this
# file — so moving them out (as 2.10.0 briefly did) turns changing that line
# into "ModuleNotFoundError: dropbox" at sign-in, on a deployment that was
# working. pCloud needs nothing beyond requests, which is already above.
msal>=1.28,<2          # Microsoft 365 / OneDrive / SharePoint, and Graph email
dropbox>=11.36,<13     # Dropbox

# Draws a report on screen, page by page, for the "Preview … on screen" boxes
# in Results. Not optional for the same reason as the backends above: the app
# offers those boxes to everybody, Streamlit Community Cloud installs only
# this file, and a browser will not display a PDF in an iframe — so without
# this package every hosted deployment loses the previews it advertises.
PyMuPDF>=1.24,<2

# The formatting toolbar for the body of every email the app sends. Core for
# the same reason: it is offered to every user, Streamlit Community Cloud
# installs only this file, and without it an instructor is handed the raw HTML
# of the message instead of the message.
streamlit-quill>=0.0.3,<1

# Backs off and retries an AI request that was rate-limited or timed out.
# Small, pure Python; without it a single rate limit becomes a failed draft.
tenacity>=8.2,<10
